111template <typename F, typename NRConfig>
112requires traits::isSpecialization<NewtonRaphsonWithSubsidiaryFunctionConfig, std::remove_cvref_t<NRConfig>>::value
113auto createNonlinearSolver(NRConfig&& config, F&& f) {
114 using LS = std::remove_cvref_t<NRConfig>::LinearSolver;
115 using UF = std::remove_cvref_t<NRConfig>::UpdateFunction;
116 using IDBCF = std::remove_cvref_t<NRConfig>::IDBCForceFunction;
117 auto solverFactory = []<class F2, class LS2, class UF2, class IDBCF2>(F2&& f2, LS2&& ls, UF2&& uf, IDBCF2&& if_) {
118 return std::make_shared<NewtonRaphsonWithSubsidiaryFunction<std::remove_cvref_t<F2>, std::remove_cvref_t<LS2>,
119 std::remove_cvref_t<UF2>, std::remove_cvref_t<IDBCF2>>>(
120 f2, std::forward<LS2>(ls), std::forward<UF2>(uf), std::forward<IDBCF2>(if_));
121 };
122
123 if constexpr (std::remove_cvref_t<F>::nDerivatives == 2) {
124 auto solver =
125 solverFactory(derivative(f), std::forward<NRConfig>(config).linearSolver,
126 std::forward<NRConfig>(config).updateFunction, std::forward<NRConfig>(config).idbcForceFunction);
127 solver->setup(config.parameters);
128 return solver;
129 } else {
130 static_assert(std::remove_cvref_t<F>::nDerivatives >= 1,
131 "The number of derivatives in the differentiable function have to be more than 0");
132 auto solver =
133 solverFactory(f, std::forward<NRConfig>(config).linearSolver, std::forward<NRConfig>(config).updateFunction,
134 std::forward<NRConfig>(config).idbcForceFunction);
135
136 solver->setup(std::forward<NRConfig>(config).parameters);
137 return solver;
138 }
139}

153class NewtonRaphsonWithSubsidiaryFunction : public NonlinearSolverBase<F>

155public:
156 using Settings = NewtonRaphsonWithSubsidiaryFunctionSettings;

178 explicit NewtonRaphsonWithSubsidiaryFunction(const DifferentiableFunction& residual, LS2&& linearSolver = {},
179 UF2&& updateFunction = {}, IDBCF2&& idbcForceFunction = {})
180 : residualFunction_{residual},
181 jacobianFunction_{derivative(residualFunction_)},
182 linearSolver_{std::forward<LS2>(linearSolver)},
183 updateFunction_{std::forward<UF2>(updateFunction)},
184 idbcForceFunction_{std::forward<IDBCF2>(idbcForceFunction)} {}

203 template <typename SubsidiaryType>
204 [[nodiscard(
205 "The solve method returns information of the solution process. You should store this information and check if "
206 "it was successful")]] NonLinearSolverInformation
207 solve(Domain& req, SubsidiaryType&& subsidiaryFunction, SubsidiaryArgs& subsidiaryArgs) {
208 using enum NonLinearSolverMessages;
209
210 Ikarus::NonLinearSolverInformation solverInformation{};
211 auto state = typename NewtonRaphsonWithSubsidiaryFunction::State{
212 .domain = req, .correction = correction_, .information = solverInformation};
213 this->notify(INIT, state);
214
215 solverInformation.success = true;
216
217 auto& lambda = req.parameter();
218 auto& x = req.globalSolution();
219
224
225 auto lambdaDummy = lambda;
226 auto DDummy = x;
227 x.setZero();
228 lambda = 1.0;
229 decltype(auto) rx = residualFunction_(req);
230 const auto Fext0 = (-rx).eval(); // dRdlambda(lambda)
231 lambda = lambdaDummy;
232 x = DDummy;
233
234 rx = residualFunction_(req);
235 decltype(auto) Ax = jacobianFunction_(req);
236
237 Eigen::VectorXd deltaD;
238 deltaD.resizeLike(rx);
239 deltaD.setZero();
240
241 subsidiaryArgs.dfdDD.resizeLike(Fext0);
242
243 subsidiaryFunction(req, *this, subsidiaryArgs);
244 auto rNorm = sqrt(rx.dot(rx));
245 solverInformation.residualNorm = static_cast<double>(rNorm);
246 decltype(rNorm) dNorm;
247 int iter{0};
248 if constexpr (isLinearSolver)
249 linearSolver_.analyzePattern(Ax);
250
251 Eigen::MatrixX2<double> residual2d, sol2d;
252
254 while (rNorm > settings_.tol && iter < settings_.maxIter) {
255 this->notify(ITERATION_STARTED, state);
256
258 residual2d.resize(rx.rows(), 2);
259 sol2d.resize(rx.rows(), 2);
260 if constexpr (not std::same_as<IDBCForceFunction, utils::IDBCForceDefault>)
261 residual2d << -rx, Fext0 - (idbcForceFunction_(req));
262 else
263 residual2d << -rx, Fext0;
264
265 if constexpr (isLinearSolver) {
266 linearSolver_.factorize(Ax);
267 linearSolver_.solve(sol2d, residual2d);
268 } else {
269 sol2d = linearSolver_(residual2d, Ax);
270 }
271
272 const auto& deltaDR = sol2d.col(0).eval();
273 const auto& deltaDL = sol2d.col(1).eval();
274
275 const double deltalambda = (-subsidiaryArgs.f - subsidiaryArgs.dfdDD.dot(deltaDR)) /
276 (subsidiaryArgs.dfdDD.dot(deltaDL) + subsidiaryArgs.dfdDlambda);
277 deltaD = deltaDR + deltalambda * deltaDL;
278 correction_ = deltaD;
279
280 this->notify(CORRECTION_UPDATED, state);
281
282 updateFunction_(x, deltaD);
283 subsidiaryArgs.DD += deltaD;
284
285 lambda += deltalambda;
286 subsidiaryArgs.Dlambda += deltalambda;
287
288 if constexpr (not std::same_as<IDBCForceFunction, utils::IDBCForceDefault>)
289 req.syncParameterAndGlobalSolution(updateFunction_);
290
291 dNorm = sqrt(deltaD.dot(deltaD) + deltalambda * deltalambda);
292 solverInformation.correctionNorm = static_cast<double>(dNorm);
293
294 rx = residualFunction_(req);
295 Ax = jacobianFunction_(req);
296 subsidiaryFunction(req, *this, subsidiaryArgs);
297
298 rNorm = sqrt(rx.dot(rx) + subsidiaryArgs.f * subsidiaryArgs.f);
299 solverInformation.residualNorm = static_cast<double>(rNorm);
300
301 this->notify(SOLUTION_CHANGED, state);
302 this->notify(CORRECTIONNORM_UPDATED, state);
303 this->notify(RESIDUALNORM_UPDATED, state);
304
305 ++iter;
306 solverInformation.iterations = iter;
307 this->notify(ITERATION_ENDED, state);
308 }
309
310 if (iter == settings_.maxIter)
311 solverInformation.success = false;
312 solverInformation.iterations = iter;
313 if (solverInformation.success)
314 this->notify(FINISHED_SUCESSFULLY, state);
315 return solverInformation;
316 }
