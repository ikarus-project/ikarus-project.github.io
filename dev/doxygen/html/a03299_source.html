41template <typename MAT, typename S>
42consteval bool hasCorrectSize() {
43 if constexpr (Concepts::EigenVector6<S> or Concepts::EigenMatrix33<S>)
44 return true;
45 if constexpr (MAT::isReduced and Concepts::EigenVector<S>) {
46 return S::RowsAtCompileTime == MAT::freeStrains;
47 } else
48 return false;
49}

213 const auto S = Impl::maybeFromVoigt(Sraw.derived(), false).eval();

215 auto [D, Eraw] = [&]() {
