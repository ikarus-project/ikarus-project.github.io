143 template <typename Derived>
144 auto materialInversionImpl(const Eigen::MatrixBase<Derived>& Sraw) const {
145 static_assert(Concepts::EigenMatrix22<decltype(Sraw)>);
146 // Enlarge S
147 auto S = Eigen::Matrix<typename Derived::Scalar, 3, 3>::Zero().eval();
148 S.template block<2, 2>(0, 0) = Sraw;
149
150 auto [D, E] = matImpl_.template materialInversion<Underlying::strainTag, true>(S);
151
152 // Reduce D and E again
153 auto Dred = reduceMatrix(D, fixedVoigtIndices);
154 auto Ered = removeCol(E, fixedVoigtIndices);
155
156 return std::make_pair(Dred, Ered);
157 }

165 template <typename Derived>
166 void initUnknownStrains(Eigen::MatrixBase<Derived>& E) const {
167 for (size_t i = 0; i < fixedPairs.size(); ++i) {
168 ScalarType initialVal = E(fixedPairs[i].row, fixedPairs[i].col);
169 if constexpr (strainTag == StrainTags::deformationGradient or strainTag == StrainTags::rightCauchyGreenTensor) {
170 if (Dune::FloatCmp::eq(initialVal, ScalarType(0.0)) and (fixedPairs[i].row == fixedPairs[i].col))
171 initialVal = ScalarType(1.0);
172 }
173 if (fixedPairs[i].row != fixedPairs[i].col)
174 initialVal = ScalarType(0.0);
175 E(fixedPairs[i].row, fixedPairs[i].col) = E(fixedPairs[i].col, fixedPairs[i].row) = initialVal;
176 }
177 }

185 template <typename Derived>
186 auto reduceStress(const Eigen::MatrixBase<Derived>& Eraw) const {
187 auto E = Impl::maybeFromVoigt(Eraw);
188 initUnknownStrains(E);
189
190 std::array<size_t, fixedDiagonalVoigtIndicesSize> fixedDiagonalVoigtIndices;
191 for (size_t ri = 0; auto i : fixedVoigtIndices) {
192 auto indexPair = fromVoigt(i);
193 if (indexPair[0] == indexPair[1])
194 fixedDiagonalVoigtIndices[ri++] = i;
195 }
196
197 auto f = [&](const auto&) {
198 auto S = matImpl_.template stresses<Underlying::strainTag, true>(E);
199 return S(fixedDiagonalVoigtIndices).eval();
200 };
201 auto df = [&](const auto&) {
202 auto moduli = (matImpl_.template tangentModuli<Underlying::strainTag, true>(E)).eval();
203 return (moduli(fixedDiagonalVoigtIndices, fixedDiagonalVoigtIndices) / Underlying::derivativeFactor).eval();
204 };
205
206 auto Er = E(fixedDiagonalVoigtIndices, fixedDiagonalVoigtIndices).eval().template cast<ScalarType>();
207
208 Er.setZero();
209 auto diffFunction = Ikarus::makeDifferentiableFunction(functions(f, df), Er);
210
211 auto linearSolver = [](auto& r, auto& A) { return (A.inverse() * r).eval(); };
212 auto updateFunction = [&](auto&, const auto& ecomps) {
213 for (int ri = 0; auto i : fixedDiagonalVoigtIndices) {
214 auto indexPair = fromVoigt(i);
215 E(indexPair[0], indexPair[1]) += ecomps(ri++);
216 }
217 };
218
219 int minIter = isAutoDiff ? 1 : 0;
220 NewtonRaphsonConfig nrs({.tol = tol_, .maxIter = 100, .minIter = minIter}, linearSolver, updateFunction);
221
222 auto nr = createNonlinearSolver(std::move(nrs), diffFunction);
223 if (!static_cast<bool>(nr->solve(Er)))
224 DUNE_THROW(Dune::MathError, "The stress reduction of material " << nameImpl() << " was unsuccessful\n"
225 << "The strains are\n"
226 << E << "\n The stresses are\n"
227 << f(Er));
228 return std::make_pair(diffFunction, E);
229 }

232 double tol_{};
