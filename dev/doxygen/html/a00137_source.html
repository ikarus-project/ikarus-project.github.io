63template <typename F, typename NRConfig>

65auto createNonlinearSolver(NRConfig&& config, F&& f) {
66 using LS = std::remove_cvref_t<NRConfig>::LinearSolver;
67 using UF = std::remove_cvref_t<NRConfig>::UpdateFunction;
68 auto solverFactory = []<class F2, class LS2, class UF2>(F2&& f2, LS2&& ls, UF2&& uf) {
69 return std::make_shared<NewtonRaphsonWithSubsidiaryFunction<std::remove_cvref_t<F2>, std::remove_cvref_t<LS2>,
70 std::remove_cvref_t<UF2>>>(f2, std::forward<LS2>(ls),
71 std::forward<UF2>(uf));
72 };
73
74 if constexpr (std::remove_cvref_t<F>::nDerivatives == 2) {
75 auto solver = solverFactory(derivative(f), std::forward<NRConfig>(config).linearSolver,
76 std::forward<NRConfig>(config).updateFunction);
77 solver->setup(config.parameters);
78 return solver;
79 } else {
80 static_assert(std::remove_cvref_t<F>::nDerivatives >= 1,
81 "The number of derivatives in the differentiable function have to be more than 0");
82 auto solver =
83 solverFactory(f, std::forward<NRConfig>(config).linearSolver, std::forward<NRConfig>(config).updateFunction);
84 ;
85
86 solver->setup(std::forward<NRConfig>(config).parameters);
87 return solver;
88 }

149 template <typename SubsidiaryType>
150 [[nodiscard(
151 "The solve method returns information of the solution process. You should store this information and check if "
152 "it was successful")]] NonLinearSolverInformation
153 solve(Domain& req, SubsidiaryType&& subsidiaryFunction, SubsidiaryArgs& subsidiaryArgs) {
154 this->notify(NonLinearSolverMessages::INIT);
155
157 Ikarus::NonLinearSolverInformation solverInformation;
158 solverInformation.success = true;
159
160 auto& lambda = req.parameter();
161 auto& x = req.globalSolution();
162
166
167 auto lambdaDummy = lambda;
168 auto DDummy = x;
169 x.setZero();
170 lambda = 1.0;
171 decltype(auto) rx = residualFunction_(req);
172 const auto Fext0 = (-rx).eval(); // dRdlambda(lambda)
173 lambda = lambdaDummy;
174 x = DDummy;
175
176 rx = residualFunction_(req);
177 decltype(auto) Ax = jacobianFunction_(req);
178
179 Eigen::VectorXd deltaD;
180 deltaD.resizeLike(rx);
181 deltaD.setZero();
182
183 subsidiaryArgs.dfdDD.resizeLike(Fext0);
184
185 subsidiaryFunction(subsidiaryArgs);
186 auto rNorm = sqrt(rx.dot(rx));
187 decltype(rNorm) dNorm;
188 int iter{0};
189 if constexpr (isLinearSolver)
190 linearSolver_.analyzePattern(Ax);
191
192 Eigen::MatrixX2<double> residual2d, sol2d;
193
195 while (rNorm > settings_.tol && iter < settings_.maxIter) {
196 this->notify(NonLinearSolverMessages::ITERATION_STARTED);
197
199 residual2d.resize(rx.rows(), 2);
200 residual2d << -rx, Fext0;
201 sol2d.resize(rx.rows(), 2);
202
203 if constexpr (isLinearSolver) {
204 linearSolver_.factorize(Ax);
205 linearSolver_.solve(sol2d, residual2d);
206 } else {
207 sol2d = linearSolver_(residual2d, Ax);
208 }
209
210 subsidiaryFunction(subsidiaryArgs);
211
212 const double deltalambda = (-subsidiaryArgs.f - subsidiaryArgs.dfdDD.dot(sol2d.col(0))) /
213 (subsidiaryArgs.dfdDD.dot(sol2d.col(1)) + subsidiaryArgs.dfdDlambda);
214 deltaD = sol2d.col(0) + deltalambda * sol2d.col(1);
215
216 updateFunction_(x, deltaD);
217 updateFunction_(subsidiaryArgs.DD, deltaD);
218
219 lambda += deltalambda;
220 subsidiaryArgs.Dlambda += deltalambda;
221
222 dNorm = sqrt(deltaD.dot(deltaD) + deltalambda * deltalambda);
223 rx = residualFunction_(req);
224 Ax = jacobianFunction_(req);
225 rNorm = sqrt(rx.dot(rx) + subsidiaryArgs.f * subsidiaryArgs.f);
226
227 this->notify(NonLinearSolverMessages::SOLUTION_CHANGED, static_cast<double>(lambda));
228 this->notify(NonLinearSolverMessages::CORRECTIONNORM_UPDATED, static_cast<double>(dNorm));
229 this->notify(NonLinearSolverMessages::RESIDUALNORM_UPDATED, static_cast<double>(rNorm));
230 this->notify(NonLinearSolverMessages::ITERATION_ENDED);
231
232 ++iter;
233 }
234
235 if (iter == settings_.maxIter)
236 solverInformation.success = false;
237 solverInformation.iterations = iter;
238 solverInformation.residualNorm = rNorm;
239 solverInformation.correctionNorm = dNorm;
240 if (solverInformation.success)
241 this->notify(NonLinearSolverMessages::FINISHED_SUCESSFULLY, iter);
242
243 return solverInformation;
244 }

250 auto& residual() { return residualFunction_; }

252private:
253 DifferentiableFunction residualFunction_;
254 typename DifferentiableFunction::Derivative jacobianFunction_;
255 std::remove_cvref_t<CorrectionType> correction_;
256 LS linearSolver_;
257 UF updateFunction_;
258 Settings settings_;
259};
