58template <typename GEO>
59struct S5 : SX<GEO, 5>
60{
61 using Base = SX<GEO, 5>;
62 static constexpr int myDim = Base::myDim;
63 static constexpr int stressSize = Base::stressSize;
64 static constexpr int assumedStressSize = Base::assumedStressSize;
65 using AnsatzType = typename Base::AnsatzType;
66 using HType = typename Base::HType;
67
68 S5() = default;
69 explicit S5(const GEO& geo)
70 : Base(geo) {}
71
77 AnsatzType operator()(const Dune::FieldVector<double, myDim>& quadPos) const {
78 AnsatzType P;
79 P.setZero(stressSize, assumedStressSize);
80 const double xi = 2.0 * quadPos[0] - 1.0;
81 const double eta = 2.0 * quadPos[1] - 1.0;
82 P(0, 0) = 1.0;
83 P(1, 1) = 1.0;
84 P(2, 2) = 1.0;
85 P(0, 3) = eta;
86 P(1, 4) = xi;
87
88 return this->T0_ * P;
89 }
90};

116 AnsatzType operator()(const Dune::FieldVector<double, myDim>& quadPos) const {
117 AnsatzType P;
118 P.setZero();
119 const double xi = 2.0 * quadPos[0] - 1.0;
120 const double eta = 2.0 * quadPos[1] - 1.0;
121 const double zeta = 2.0 * quadPos[2] - 1.0;
122
123 // ξξ = (1.0, eta, zeta, eta*zeta) → row 0:
124 P(0, 0) = 1.0;
125 P(0, 1) = eta;
126 P(0, 2) = zeta;
127 P(0, 3) = eta * zeta;
128
129 // ηη = (1.0, xi, zeta, xi*zeta) → row 1:
130 P(1, 4) = 1.0;
131 P(1, 5) = xi;
132 P(1, 6) = zeta;
133 P(1, 7) = xi * zeta;
134
135 // ζζ = (1.0, xi, eta, xi*eta) → row 2:
136 P(2, 8) = 1.0;
137 P(2, 9) = xi;
138 P(2, 10) = eta;
139 P(2, 11) = xi * eta;
140
141 // ηζ = (1, xi) → row 3:
142 P(3, 12) = 1.0;
143 P(3, 13) = xi;
144
145 // ξζ = (1.0, eta) → row 4:
146 P(4, 14) = 1.0;
147 P(4, 15) = eta;
148
149 // ξη = (1.0, zeta) → row 5:
150 P(5, 16) = 1.0;
151 P(5, 17) = zeta;
152
153 return this->T0_ * P;
154 }
155};
